name: Ephemeral AI Chatbot on: workflow_dispatch: inputs: model_size: description: 'Select AI Model' required: true default: 'llama3' type: choice options: - llama3 - mistral - phi3:medium - qwen:0.5b jobs: run-chatbot: runs-on: ubuntu-latest timeout-minutes: 60 permissions: contents: write steps: - name: Checkout Repository uses: actions/checkout@v4 - name: Start Ollama Service run: | curl -fsSL https://ollama.com/install.sh | sh export OLLAMA_HOST=0.0.0.0:11434 export OLLAMA_ORIGINS=* nohup ollama serve > ollama.log 2>&1 & sleep 10 - name: Pull Selected AI Model run: | ollama pull ${{ github.event.inputs.model_size }} - name: Start Open WebUI (GUI) run: | # Install Open WebUI requirements pip install open-webui # Start Open WebUI in background on port 3000 export OLLAMA_BASE_URL=http://127.0.0.1:11434 export PORT=3000 nohup open-webui serve > open-webui.log 2>&1 & echo "Waiting for WebUI to initialize..." sleep 20 - name: Expose via Cloudflare & Update GitHub Page run: | curl -L --output cloudflared.deb https://github.com/cloudflare/cloudflared/releases/latest/download/cloudflared-linux-amd64.deb sudo dpkg -i cloudflared.deb # Tunnel to port 3000 cloudflared tunnel --url http://localhost:3000 > tunnel.log 2>&1 & sleep 10 TUNNEL_URL=$(grep -oE "https://[a-zA-Z0-9.-]+\.trycloudflare\.com" tunnel.log | head -n 1) # Update index.html sed -i "s|https://mpegs-nano-newest-jar.trycloudflare.com|$TUNNEL_URL|g" index.html git config --global user.name "github-actions[bot]" git config --global user.email "github-actions[bot]@users.noreply.github.com" git add index.html git commit -m "Update live GUI URL: $TUNNEL_URL" git push echo "Chatbot GUI is live at: $TUNNEL_URL" sleep 3500
